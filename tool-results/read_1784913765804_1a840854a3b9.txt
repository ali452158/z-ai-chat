    83→function extractCodeBlocks(content: string) {
    84→  const regex = /```(\w+)?\n([\s\S]*?)```/g
    85→  const blocks: { language: string; code: string }[] = []
    86→  let match
    87→  while ((match = regex.exec(content)) !== null) {
    88→    blocks.push({
    89→      language: match[1] || 'text',
    90→      code: match[2].trim(),
    91→    })
    92→  }
    93→  return blocks
    94→}
