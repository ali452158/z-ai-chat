    82→function extractCodeBlocks(content: string) {
    83→  const regex = /```(\w+)?\n([\s\S]*?)```/g
    84→  const blocks: { language: string; code: string }[] = []
    85→  let match
    86→  while ((match = regex.exec(content)) !== null) {
    87→    blocks.push({
    88→      language: match[1] || 'text',
    89→      code: match[2].trim(),
    90→    })
    91→  }
    92→  return blocks
    93→}
