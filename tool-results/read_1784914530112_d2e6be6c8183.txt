    82→    82→function extractCodeBlocks(content: string) {
    83→    83→  const regex = /```(\w+)?\n([\s\S]*?)```/g
    84→    84→  const blocks: { language: string; code: string }[] = []
    85→    85→  let match
    86→    86→  while ((match = regex.exec(content)) !== null) {
    87→    87→    blocks.push({
    88→    88→      language: match[1] || 'text',
    89→    89→      code: match[2].trim(),
    90→    90→    })
    91→    91→  }
    92→    92→  return blocks
    93→    93→}
